你遇到的问题是在同一页面同时使用 Three.js 和 ModelViewer 时出现 WebGL 黑屏，核心原因确实是两者争夺 WebGL 上下文资源以及渲染循环相互干扰导致的。ModelViewer 底层其实也依赖 Three.js，这会造成上下文冲突和渲染逻辑的相互影响。

### 问题分析与解决方案
我会为你提供一套完整的解决方案，核心思路是：
1. 为 Three.js 和 ModelViewer 分配独立的 WebGL 上下文
2. 统一管理渲染循环，避免相互干扰
3. 确保资源加载和销毁的正确时机

#### 完整解决方案代码
```javascript
// 全局配置 - 关键：为不同组件分配独立的 WebGL 上下文
const WEBGL_CONFIG = {
  // Three.js 专用配置
  three: {
    antialias: true,
    alpha: true,
    preserveDrawingBuffer: true,
    powerPreference: "high-performance"
  },
  // ModelViewer 专用配置
  modelViewer: {
    antialias: true,
    alpha: true,
    preserveDrawingBuffer: false
  }
};

// Three.js 场景管理器
class ThreeSceneManager {
  constructor(containerId) {
    this.container = document.getElementById(containerId);
    this.scene = null;
    this.camera = null;
    this.renderer = null;
    this.animationId = null;
    this.isRendering = false;
  }

  // 初始化 Three.js 场景
  init() {
    // 1. 创建独立的渲染器，关键是传递专属的 WebGL 配置
    this.renderer = new THREE.WebGLRenderer({
      ...WEBGL_CONFIG.three,
      // 关键：明确指定不共享上下文
      context: null
    });
    this.renderer.setSize(this.container.clientWidth, this.container.clientHeight);
    this.renderer.setPixelRatio(window.devicePixelRatio);
    this.container.appendChild(this.renderer.domElement);

    // 2. 创建场景和相机
    this.scene = new THREE.Scene();
    this.scene.background = new THREE.Color(0x87ceeb); // 天空蓝，便于区分

    this.camera = new THREE.PerspectiveCamera(
      75,
      this.container.clientWidth / this.container.clientHeight,
      0.1,
      1000
    );
    this.camera.position.z = 5;

    // 3. 添加测试内容
    const geometry = new THREE.BoxGeometry(1, 1, 1);
    const material = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
    const cube = new THREE.Mesh(geometry, material);
    this.scene.add(cube);

    // 4. 启动统一的渲染循环
    this.startRenderLoop();

    // 5. 监听窗口大小变化
    window.addEventListener('resize', () => this.onWindowResize());

    return this;
  }

  // 统一的渲染循环 - 关键：避免多个 requestAnimationFrame 冲突
  startRenderLoop() {
    if (this.isRendering) return;
    
    this.isRendering = true;
    const render = () => {
      this.animationId = requestAnimationFrame(render);
      this.renderer.render(this.scene, this.camera);
    };
    render();
  }

  // 停止 Three.js 渲染循环
  stopRenderLoop() {
    if (this.animationId) {
      cancelAnimationFrame(this.animationId);
      this.animationId = null;
    }
    this.isRendering = false;
  }

  // 窗口大小适配
  onWindowResize() {
    this.camera.aspect = this.container.clientWidth / this.container.clientHeight;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(this.container.clientWidth, this.container.clientHeight);
  }

  // 销毁 Three.js 资源，释放 WebGL 上下文
  dispose() {
    this.stopRenderLoop();
    
    // 清理场景
    while (this.scene.children.length) {
      const child = this.scene.children[0];
      if (child.geometry) child.geometry.dispose();
      if (child.material) {
        if (Array.isArray(child.material)) {
          child.material.forEach(mat => mat.dispose());
        } else {
          child.material.dispose();
        }
      }
      this.scene.remove(child);
    }

    // 释放渲染器资源
    this.renderer.dispose();
    this.renderer.forceContextLoss();
    this.container.removeChild(this.renderer.domElement);
    
    // 清空引用
    this.scene = null;
    this.camera = null;
    this.renderer = null;
  }
}

// ModelViewer 管理器
class ModelViewerManager {
  constructor(containerId) {
    this.container = document.getElementById(containerId);
    this.modelViewer = null;
  }

  // 初始化 ModelViewer
  init(modelUrl) {
    // 1. 创建 model-viewer 元素
    this.modelViewer = document.createElement('model-viewer');
    this.modelViewer.setAttribute('src', modelUrl);
    this.modelViewer.setAttribute('alt', '3D Model');
    this.modelViewer.setAttribute('auto-rotate', 'true');
    this.modelViewer.setAttribute('camera-controls', 'true');
    this.modelViewer.style.width = '100%';
    this.modelViewer.style.height = '100%';
    
    // 2. 关键配置：为 ModelViewer 指定独立的 WebGL 上下文
    this.modelViewer.webglPreferences = {
      ...WEBGL_CONFIG.modelViewer
    };

    // 3. 监听加载事件，避免加载过程中干扰 Three.js
    this.modelViewer.addEventListener('load', () => {
      console.log('Model loaded successfully');
      // 确保 ModelViewer 使用自己的渲染循环
      this.modelViewer.pauseRendering = false;
    });

    // 4. 监听错误
    this.modelViewer.addEventListener('error', (e) => {
      console.error('ModelViewer error:', e);
    });

    this.container.appendChild(this.modelViewer);

    return this;
  }

  // 暂停 ModelViewer 渲染
  pause() {
    if (this.modelViewer) {
      this.modelViewer.pauseRendering = true;
    }
  }

  // 恢复 ModelViewer 渲染
  resume() {
    if (this.modelViewer) {
      this.modelViewer.pauseRendering = false;
    }
  }

  // 销毁 ModelViewer
  dispose() {
    if (this.modelViewer) {
      this.pause();
      this.container.removeChild(this.modelViewer);
      this.modelViewer = null;
    }
  }
}

// 页面初始化 - 确保资源加载顺序和上下文隔离
document.addEventListener('DOMContentLoaded', async () => {
  try {
    // 1. 先初始化 Three.js 场景
    const threeManager = new ThreeSceneManager('three-container');
    threeManager.init();

    // 2. 延迟初始化 ModelViewer，避免上下文竞争
    setTimeout(async () => {
      const modelViewerManager = new ModelViewerManager('modelviewer-container');
      // 替换为你的 MDX 模型路径
      modelViewerManager.init('path/to/your/model.mdx');
    }, 1000);

    // 3. 全局清理（页面卸载时）
    window.addEventListener('beforeunload', () => {
      threeManager.dispose();
      // 如果需要，也销毁 ModelViewer
      // modelViewerManager.dispose();
    });
  } catch (error) {
    console.error('Initialization error:', error);
  }
});
```

### 关键技术点解释
1. **独立 WebGL 上下文配置**
   - 为 Three.js 和 ModelViewer 分别配置不同的 `WebGLRenderer` 参数，特别是 `preserveDrawingBuffer` 和 `context` 属性
   - ModelViewer 通过 `webglPreferences` 配置专属的 WebGL 参数，避免与 Three.js 共享上下文

2. **统一渲染循环管理**
   - Three.js 使用独立的渲染循环，并通过 `isRendering` 标志防止重复创建
   - ModelViewer 利用自身的 `pauseRendering` API 控制渲染状态，避免多个 `requestAnimationFrame` 冲突
   - 延迟初始化 ModelViewer（1秒延迟），避免初始化时的资源竞争

3. **资源生命周期管理**
   - 实现完整的 `dispose` 方法，释放几何、材质、渲染器等资源
   - 使用 `forceContextLoss()` 强制释放 WebGL 上下文，避免内存泄漏
   - 页面卸载时清理所有资源，防止上下文残留

4. **调试建议**
   - 打开 Chrome DevTools → More tools → WebGL Inspector 查看上下文状态
   - 添加错误监听，捕获潜在的 WebGL 错误：
     ```javascript
     renderer.context.getExtension('WEBGL_debug_renderer_info');
     renderer.domElement.addEventListener('webglcontextlost', (e) => {
       console.error('WebGL context lost:', e);
       e.preventDefault();
     });
     ```

### 总结
1. **核心问题**：Three.js 和 ModelViewer 共享同一页面时会争夺 WebGL 上下文，且各自的渲染循环会相互干扰，导致黑屏。
2. **关键解决策略**：为两者分配独立的 WebGL 上下文配置，统一管理渲染循环，控制初始化顺序和资源生命周期。
3. **最佳实践**：使用封装的管理器类分别控制 Three.js 和 ModelViewer，实现资源的按需创建和销毁，避免上下文冲突。

如果问题仍然存在，可以尝试：
- 降低 ModelViewer 的渲染精度（关闭抗锯齿）
- 确保两者使用相同版本的 Three.js 核心
- 使用 iframe 彻底隔离两个组件的渲染环境
